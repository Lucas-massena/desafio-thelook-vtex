Código BigQuery Auditável (Trilha Técnica)
Pipeline SQL de Backtest e Treinamento do Modelo
BigQuery ML 
-- BACKTEST AUDITADO: Simulação do dia 17 do mês anterior
WITH historico_dia17 AS (
  SELECT 
    AVG(sale_price) * 13 AS projecao_media_28d
  FROM `thelook_ecommerce.order_items`
  WHERE status NOT IN ('Cancelled', 'Returned')
    AND DATE(created_at) BETWEEN DATE_SUB(DATE '2026-09-17', INTERVAL 28 DAY) AND DATE '2026-09-17'
)
-- INFERÊNCIA TIMESFM / AI.FORECAST
SELECT 
  forecast_timestamp,
  ROUND(forecast_value, 2) AS projecao_timesfm,
  ROUND(prediction_interval_lower_bound, 2) AS pessimista,
  ROUND(prediction_interval_upper_bound, 2) AS otimista
FROM ML.FORECAST(MODEL `vtex_workshop.timesfm_thelook`,
  STRUCT(13 AS horizon, 0.9 AS confidence_level));
